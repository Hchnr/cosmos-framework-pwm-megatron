def named_running(name):
    for item in psutil.process_iter(['cmdline']):
        args = item.info['cmdline'] or []
        if any(arg.endswith('/monitor_tp_pp_job.py') for arg in args):
            continue
        if 'job.name=' + name in args:
            return True
        if '--job-name' in args and args.index('--job-name') + 1 < len(args):
            if args[args.index('--job-name') + 1] == name:
                return True
    return False

def profile_job(name, command, environment, profile_name):
    assert not named_running(name)
    profiling = ['nsys', 'profile', '--trace=cuda,nvtx,nccl', '--sample=none', '--cpuctxsw=none',
                 '--capture-range=cudaProfilerApi', '--capture-range-end=stop-shutdown', '--kill=none',
                 '--wait=primary', '--force-overwrite=false',
                 '--output=/opt/cosmos-tp-pp-profiles/' + profile_name, *command]
    with open('/tmp/' + profile_name + '.log', 'x') as log:
        process = subprocess.Popen(profiling, env=environment, stdout=log, stderr=subprocess.STDOUT)
        print(json.dumps({'launched_job':name,'nsys_pid':process.pid,'command':command}), flush=True)
        code=process.wait()
        print(json.dumps({'job':name,'nsys_exit_code':code,'job_still_running':named_running(name)}), flush=True)
        while named_running(name):
            time.sleep(30)
